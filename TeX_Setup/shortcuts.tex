\newcommand{\cyclevertices}[2][1]{  % [radius]{n}: the n vertices of a regular n-gon
    \foreach \i in {1, ..., #2}{
        \pgfmathsetmacro{\angthis}{90 + 360 * (\i - 1) / #2}
        \fill[black] (\angthis:#1) circle (2pt);
    }
}

\newcommand{\stepedges}[3][1]{  % [radius]{n}{step}: the edges joining each vertex of a regular n-gon to the one `step' further round
    \foreach \i in {1, ..., #2}{
        \pgfmathsetmacro{\angthis}{90 + 360 * (\i - 1) / #2}
        \pgfmathsetmacro{\angnext}{90 + 360 * (\i - 1 + #3) / #2}
        \draw[thick] (\angthis:#1) -- (\angnext:#1);
    }
}

\newcommand{\cyclegraph}[2][1]{  % [radius]{n}: the cycle graph on n vertices, drawn as a regular n-gon
    \stepedges[#1]{#2}{1}
    \cyclevertices[#1]{#2}
}

\newcommand{\fanoplane}[1][1.6]{  % [radius]: the Fano plane, drawn as a triangle with its medians and the circle through its edge midpoints
    \pgfmathsetmacro{\rmid}{0.5 * #1}
    \foreach \i in {0, 1, 2}{
        \pgfmathsetmacro{\angcorner}{90 + 120 * \i}
        \pgfmathsetmacro{\angnext}{210 + 120 * \i}
        \pgfmathsetmacro{\angopp}{270 + 120 * \i}
        \draw[thick] (\angcorner:#1) -- (\angnext:#1);
        \draw[thick] (\angcorner:#1) -- (\angopp:\rmid);
    }
    \draw[thick] (0, 0) circle (\rmid);
    \foreach \i in {0, 1, 2}{
        \pgfmathsetmacro{\angcorner}{90 + 120 * \i}
        \pgfmathsetmacro{\angmid}{150 + 120 * \i}
        \fill[black] (\angcorner:#1) circle (2pt);
        \fill[black] (\angmid:\rmid) circle (2pt);
    }
    \fill[black] (0, 0) circle (2pt);
}

\newcommand{\stargraph}[2][1]{  % [radius]{n}: the star graph with n edges, drawn as a hub with n spokes

\newcommand{\cyclearc}[4][1]{  % [radius]{n}{k}{s}: the n-cycle as n dots on a circle, with the k-interval running from vertex s highlighted
    \pgfmathtruncatemacro{\klast}{#3 - 1}
    \pgfmathsetmacro{\angfrom}{90 + 360 * (#4 - 1) / #2}
    \pgfmathsetmacro{\angto}{90 + 360 * (#4 - 2 + #3) / #2}
    \draw[gray!50] (0, 0) circle (#1);
    \draw[very thick, magenta] (\angfrom:#1) arc (\angfrom:\angto:#1);
    \foreach \i in {1, ..., #2}{
        \pgfmathsetmacro{\angthis}{90 + 360 * (\i - 1) / #2}
        \fill[black] (\angthis:#1) circle (2pt);
    }
    \foreach \j in {0, ..., \klast}{
        \pgfmathsetmacro{\angdot}{90 + 360 * (#4 - 1 + \j) / #2}
        \fill[magenta] (\angdot:#1) circle (3pt);
    }
}

\newcommand{\cyclefan}[4][1]{  % [radius]{n}{k}{s}: the n-cycle, with the k intervals through vertex s drawn as nested arcs outside it
    \pgfmathtruncatemacro{\klast}{#3 - 1}
    \draw[gray!50] (0, 0) circle (#1);
    \foreach \i in {1, ..., #2}{
        \pgfmathsetmacro{\angthis}{90 + 360 * (\i - 1) / #2}
        \fill[black] (\angthis:#1) circle (2pt);
    }
    \foreach \j in {0, ..., \klast}{
        \pgfmathsetmacro{\rarc}{#1 + 0.17 * (\j + 1)}
        \pgfmathsetmacro{\angfrom}{90 + 360 * (#4 - #3 + \j) / #2}
        \pgfmathsetmacro{\angto}{90 + 360 * (#4 - 1 + \j) / #2}
        \draw[thick, magenta] (\angfrom:\rarc) arc (\angfrom:\angto:\rarc);
    }
    \pgfmathsetmacro{\angx}{90 + 360 * (#4 - 1) / #2}
    \fill[magenta] (\angx:#1) circle (3pt);
}

\newcommand{\bipartiteparts}[3][0.9]{  % [spacing]{m}{n}: the two parts of a bipartite graph, as columns of m and n vertices named (A1), ..., (Am) and (B1), ..., (Bn), ready to be joined by \draw (A1) -- (B2);
    \foreach \i in {1, ..., #2}{
        \pgfmathsetmacro{\yleft}{#1 * ((#2 + 1) / 2 - \i)}
        \node[circle, fill = black, inner sep = 1.5pt] (A\i) at (0, \yleft) {};
    }
    \foreach \j in {1, ..., #3}{
        \pgfmathsetmacro{\yright}{#1 * ((#3 + 1) / 2 - \j)}
        \node[circle, fill = black, inner sep = 1.5pt] (B\j) at (3, \yright) {};
    }
}
