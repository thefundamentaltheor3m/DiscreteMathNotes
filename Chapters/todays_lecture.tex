\begin{boxdefinition}[Complete Uniform Hypergraph]
    Define the \textbf{complete $r$-uniform hypergraph on $n$ vertices}, denoted $K_{(r)}(n)$, to be the $r$-hypergraph on $n$ vertices with all possible hyper-edges.
\end{boxdefinition}

\begin{boxdefinition}[$k$-clique]
    Given a hypergraph and a colouring on that hypergraph, a \textbf{$k$-clique of some colour} is a set  of $k$ vertices such that all possible hyper-edges on that vertex set are monochromatic, of that colour. The \textbf{size} of a $k$-clique is the number of edges. % [CLAUDE] is that correct? If not, correct it.
\end{boxdefinition}

\begin{boxdefinition}[Hypergraph Ramsey Number]
    Given $r, k, \ell \in \N$, we define $R_{(r)}(k, \ell)$ to be the least $n$ such that any red/blue colouring of $K_{(r)}(n)$ contains a red $k$-c;oqie or a blue $\ell$-clique.
\end{boxdefinition}
