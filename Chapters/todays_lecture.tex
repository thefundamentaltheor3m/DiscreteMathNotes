\begin{boxtheorem}
    Any Kakeya set in $\F_q^n$ has size at least $\parenth{\frac{q}{10n}}^n$.
\end{boxtheorem}
\begin{proof}
    Suppose not. Choose a nonzero polynomial $p$ of degree at most $n\abs{K}^{\frac{1}{n}}$. Write $c_n = \parenth{\frac{1}{10n}}^n$. If $\abs{K} > c_n q^n$, then $\deg(p) \leq q - 1$. $q - 2$ that vanishes on $K$.

    For all $a \in \F_q^n \setminus \set{0}$, there is some $b$ such that the line
    \begin{align*}
        L_{a, b} = \setst{at + b}{t \in \F_q}
    \end{align*}
    is contained in $K$. We know, in particular, that
    \begin{align*}
        \phi(t) = p\of{at + b}
    \end{align*}
    is the zero polynomial.

    Let $D = \deg(p)$. Write $p = p_D + R$, where $p_D$ is the degree $D$ part of $p$ (ie, the part with all monomials having degree $D$) and $R$ is whatever's left.Observe that
    \begin{align*}
        \phi(t) = p_D(a) t^D + \cdots + \text{something}
    \end{align*}
    The fact that $\phi(t)$ vanishes tells us that $p_D(a) = 0$ for all $a \in \F_q^n \setminus \set{0}$. But also $p_D(0) = 0$. So in fact $p_D(a) = 0$ for all $a \in \F_q^n$. So by our previous lemma, $p_D = 0$, a contradiction.
\end{proof}

\begin{boxlemma}[Schwarz-Zippel]
    Let $\F$ be an arbitrary field and let $p \in \poly_d\of{\F^n}$ be nonzero. For any finite subset $S \subset \F^n$, if $s_1, \ldots, s_n \in S$ are chosen uniformly at random (with replacement), then
    \begin{align*}
        \Pr\of{p\of{s_1, \ldots, s_n} = 0} \leq \frac{d}{\abs{S}}
    \end{align*}
\end{boxlemma}

This essentially tells us that a nonzero polynomial of degree $d$ cannot vanish on more than a $\frac{d}{\abs{S}}$ fraction of any grid-set of the form
\begin{align*}
    \underbrace{S \times \cdots \times S}_{n \text{ times}}
\end{align*}

\begin{boxdefinition}[List Chromatic Number]
    For a graph $G$, define the \textbf{list chromatic number $\chi'(G)$} to be the smallest $k$ such that for any assignment of lists of size $k$ to vertices of $G$, there is a proper colouring from the list.
\end{boxdefinition}

Our strategy for finding upper-bounds will be to apply the Combinatorial Nullstellensatz to the folowing polynomial:
\begin{align}
    p\of{x_1, \ldots, x_n} = \prod_{i \sim j} \parenth{x_i - x_j}
\end{align}
where we label the vertices of the graph $\set{1, \ldots, n}$ and take the product over adjacent vertices, ie, over edges.

\begin{boxtheorem}
    For all $k \leq q - 1$,
    \begin{align*}
        \sum_{x \in \F_q} x^k = 0
    \end{align*}
\end{boxtheorem}
\begin{proof}
    Note that $x^{k} - 1$ has at most $k < q$ roots over $\F_q$. So there is certainly some $a \in \F_q$ that is \textit{not} a root of this polynomial. So $a^k \neq 1$. Since $x \mapsto ax$ is a bijection,
    \begin{align*}
        \sum_{x \in \F_q} x^k = \sum_{x \in \F_q} \parenth{ax}^k = a^k \sum_{x \in \F_q} x^k
    \end{align*}
    As $a^k \ne 1$, the only way this equality can hold is if $\sum_{x \in \F_q} x^k = 0$.
\end{proof}

We'll now prove Fermat's Little Theorem combinatorially.

\begin{boxtheorem}[Fermat's Little Theorem]
    For $p$ a prime, for all $a \in \Z$, $a^p \equiv a \pmod{p}$. Equivalently, for $p$ a prime, for all $a \in \Z$ such that $p \nmid a$, $a^{p - 1} \equiv 1 \pmod{p}$.
\end{boxtheorem}
\begin{proof}
    We know that $x \mapsto ax$ is a bijection. So
    \begin{align*}
        \prod_{x \in \F_p \setminus \set{0}} x
        = \prod_{x \in \F_p \setminus \set{0}} (ax)
        = a^{p - 1} \prod_{x \in \F_p \setminus \set{0}} x = a^{p - 1}
    \end{align*}
    in $\F_p$.
\end{proof}

Of course the corresponding result holds in $\F_q$ as well.

\begin{boxtheorem}
    For all $a \in \F_q \setminus \set{0}$, $a^{q - 1} = 1$.
\end{boxtheorem}
\begin{proof}[``Proof'']
    $\F_q$ is, by definition, the splitting field of $x^{q} - x$ over $\F_p$, where $q$ is a power of $p$. So when $a \in \F_q$ is nonzero,
    \begin{align*}
        a^q - a = 0 \iff a\parenth{a^{q - 1} - 1} = 0 \iff a^{q - 1} - 1 = 0
    \end{align*}
\end{proof}

\begin{boxtheorem}[Chevalley-Warning]
    Fix $p_1, \ldots, p_k \in \F_q\!\brac{x_1, \ldots, x_n}$ with $\sum_{i=1}^{k} \pdeg{p_i} < n$. Let $p$ be the characteristic of $\F_q$. Then
    \begin{align*}
        p \mid \#\!\setst{\bx \in \F_q}{p_1\of{\bx} = p_2\of{\bx} = \cdots = p_k\of{\bx} = 0}
    \end{align*}
\end{boxtheorem}
\begin{proof}
    Define the polynomial
    \begin{align*}
        Q(\bx) = \prod_{i=1}^{k} \parenth{1 - p_i(\bx)^{q - 1}}
    \end{align*}
    What's interesting about the way we've defined $Q$ is that
    \begin{itemize}
        \item $Q(\bx) = 0$ iff $p_i(\bx) \ne 0$ for \textit{some} $i$
        \item $Q(\bx) = 1$ iff $p_i(\bx) = 0$ for \textit{every} $i$
    \end{itemize}
    Consider the quantity
    \begin{align*}
        \F_q \ni z = \sum_{\bx \in \F_q^n} Q(\bx)
    \end{align*}
    Notice that $\pdeg{Q} < n(q - 1)$, so by previous theorem, % [CLAUDE] add cross-ref
    $z = 0$. We can write
    \begin{align*}
        z = \underbrace{1 + \cdots + 1}_{N \text{ times}}
    \end{align*}
    where $N = \#\!\setst{\bx \in \F_q}{p_1\of{\bx} = p_2\of{\bx} = \cdots = p_k\of{\bx} = 0}$. As $N = 0$ in $\F_q$, we must have $p \mid N$.
\end{proof}
