\begin{boxtheorem}
    Any Kakeya set in $\F_q^n$ has size at least $\parenth{\frac{q}{10n}}^n$.
\end{boxtheorem}
\begin{proof}
    Suppose not. Choose a nonzero polynomial $p$ of degree at most $n\abs{K}^{\frac{1}{n}}$. Write $c_n = \parenth{\frac{1}{10n}}^n$. If $\abs{K} > c_n q^n$, then $\deg(p) \leq q - 1$. $q - 2$ that vanishes on $K$.

    For all $a \in \F_q^n \setminus \set{0}$, there is some $b$ such that the line
    \begin{align*}
        L_{a, b} = \setst{at + b}{t \in \F_q}
    \end{align*}
    is contained in $K$. We know, in particular, that
    \begin{align*}
        \phi(t) = p\of{at + b}
    \end{align*}
    is the zero polynomial.

    Let $D = \deg(p)$. Write $p = p_D + R$, where $p_D$ is the degree $D$ part of $p$ (ie, the part with all monomials having degree $D$) and $R$ is whatever's left.Observe that
    \begin{align*}
        \phi(t) = p_D(a) t^D + \cdots + \text{something}
    \end{align*}
    The fact that $\phi(t)$ vanishes tells us that $p_D(a) = 0$ for all $a \in \F_q^n \setminus \set{0}$. But also $p_D(0) = 0$. So in fact $p_D(a) = 0$ for all $a \in \F_q^n$. So by our previous lemma, $p_D = 0$, a contradiction.
\end{proof}

\begin{boxlemma}[Schwarz-Zippel]
    Let $\F$ be an arbitrary field and let $p \in \poly_d\of{\F^n}$ be nonzero. For any finite subset $S \subset \F^n$, if $s_1, \ldots, s_n \in S$ are chosen uniformly at random (with replacement), then
    \begin{align*}
        \Pr\of{p\of{s_1, \ldots, s_n} = 0} \leq \frac{d}{\abs{S}}
    \end{align*}
\end{boxlemma}

This essentially tells us that a nonzero polynomial of degree $d$ cannot vanish on more than a $\frac{d}{\abs{S}}$ fraction of any grid-set of the form
\begin{align*}
    \underbrace{S \times \cdots \times S}_{n \text{ times}}
\end{align*}

Let's think about how we'd prove Schwarz-Zippel. Suppose we choose $s_1, \ldots, s_n$ randomly. View $p\of{x_1, \ldots, x_n}$ as a single-variable polynomial in $x_n$, ie, as an element of $\F\!\brac{x_1, \ldots, x_{n - 1}}\!\brac{x_n}$. Write

    = p_0\of{s_1, \ldots, s_n} + p_1\of{s_1, \ldots, s_{n-1}} s_n + p_2\of{s_1, \ldots, s_{n-1}} s_n^2 + \cdots + p_k\of{} \sorry

\begin{boxtheorem}[Combinatorial Nullstellensatz]
    Suppose $p \in \poly_D\of{\F^n}$ is a polynomial with a nonzero coefficient for $x_1^{j_1} x_2^{j_2} \cdots x_n j^{n}$. Then, for any $S_1, \ldots, S_n \ssq \F$ with $\abs{S_i} \geq j_i + 1$, we have that some $x \in S_1 \times \cdots S_n$ has $p(x) \neq 0$.
\end{boxtheorem}
\begin{remark}
    Note that the condition that the coefficient for $x_1^{j_1} x_2^{j_2} \cdots x_n j^{n}$ is nonzero implies that the total degree is at least $j_1 + \cdots + j_n$.
\end{remark}
\begin{proof}
    Fix $p$ as above. WLoG, assume $j_1 \geq 1$ (so that $\abs{S_1} \geq 2$). For $s \in \F$, write
    \begin{align*}
        p\of{x_1, \ldots, x_n} = \parenth{x_1 - s} q_s\of{x_1, \ldots, x_n} + r_s\of{x_2, \ldots, x_n}
    \end{align*}
    Notice that
    \begin{align*}

    \end{align*}

\begin{boxdefinition}[List Chromatic Number]
    For a graph $G$, define the \textbf{list chromatic number $\chi'(G)$} to be the smallest $k$ such that for any assignment of lists of size $k$ to vertices of $G$, there is a proper colouring from the list.
\end{boxdefinition}

Our strategy for finding upper-bounds will be to apply the Combinatorial Nullstellensatz to the folowing polynomial:
\begin{align}
    p\of{x_1, \ldots, x_n} = \prod_{i \sim j} \parenth{x_i - x_j}
\end{align}
where we label the vertices of the graph $\set{1, \ldots, n}$ and take the product over adjacent vertices, ie, over edges.
